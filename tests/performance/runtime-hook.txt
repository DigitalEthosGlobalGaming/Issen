window.__profile = {
 G, fx, frameLoop, startRun, spawnEnemy, enemyPos, killEnemy, startTrial, openPanel, closePanel, pause,
 state: () => ({state:G.state,panel:G.panel,enemies:G.enemies.length,runTime:G.runTime,kills:G.kills,time,particles:Object.values(fx).reduce((n,a)=>n+a.length,0),transitions:window.__sceneTransitions||0,renderedScenes:[...(window.__renderedScenes||[])],driftLeaves:presentation.environmentState.leaves.length,gustLeaves:presentation.environmentState.leaves.filter(leaf=>leaf.gust).length}),
 async configure(name) {
  window.__scenario=name;
  if(name==='scene-transitions') { window.__sceneTransitions=0; window.__nextScene=0; window.__sceneAt=performance.now()+650; }
  if(name==='cinematic-transitions') {
   cinematic.open(0); if(!cinematic.active) throw Error('Cinematic did not open');
   window.__sceneTransitions=0; window.__renderedScenes=new Set(); window.__sceneAt=performance.now()+350;
  }
  if (name==='demon') { startTrial('demon-mirror'); if(!activeTrial) throw Error('Trial did not start'); }
  else if(['combat','drift-calm','drift-gust','paused','inactive-combat','film-glitch','film-inferno','kill-effects','stress-100'].includes(name)) startRun();
  if(name.startsWith('drift-')) {
   const deadline=performance.now()+45000;
   while(cvs.dataset.sceneState==='loading') { if(performance.now()>deadline) throw Error('Drift scene did not become ready'); await new Promise(resolve=>setTimeout(resolve,20)); }
   window.__driftGustAt=0;
  }
  if(name==='paused') pause();
  if(name==='film-glitch') EQ.film='trial-glitch';
  if(name==='film-inferno') EQ.film='trial-inferno';
  if(name==='kill-effects') EQ.fx='scattered-armour';
  if(name==='stress-100') {
   const deadline=performance.now()+45000;
   while(cvs.dataset.sceneState==='loading') { if(performance.now()>deadline) throw Error('Stress scene did not become ready'); await new Promise(resolve=>setTimeout(resolve,20)); }

   window.__resetProfileRandom(); time=0;
   G.enemies=[]; G.pendingSpawns=[]; G.toSpawn=0; G.attacker=null; G.gapT=1e9;
   for(let i=0;i<100;i++) { const e=spawnEnemy(i%5,true); e.state='idle'; e.fixed={x:W*(.06+(i%10)*.098),y:H*(.22+Math.floor(i/10)*.054),h:H*.14,fog:(i%5)*.08}; e.pos=enemyPos(e); }
  }
 },
 beforeUpdate() {
  if(window.__scenario?.startsWith('drift-')) {
   foundation.run.WX.gustT=1e9;
   if(window.__scenario==='drift-gust' && performance.now()>=window.__driftGustAt) {
    presentation.gustLeaves(70);
    window.__driftGustAt=performance.now()+1500;
   }
  }
  if(window.__scenario==='cinematic-transitions') {
   const scene=cvs.dataset.rendererBackend==='demon-realm'?STAGES.length:environmentRenderer.snapshot().stage;
   const requested=Number(document.querySelector('#cinematic').dataset.scene);
   if(window.__probe.measure) {
    if(Number.isInteger(scene)) window.__renderedScenes.add(scene);
   }
   if(scene===requested && performance.now()>=window.__sceneAt) {
    cinematic.swipe('left'); window.__sceneTransitions++; window.__sceneAt=performance.now()+350;
   }
  }
  if(window.__scenario==='scene-transitions' && performance.now()>=window.__sceneAt) {
   previewStage(window.__nextScene++%(STAGES.length+1));
   window.__sceneTransitions++;
   window.__sceneAt=performance.now()+650;
  }
  if(window.__scenario==='stress-100') {G.gapT=1e9; return;}
  const e=G.attacker;
  if(e && e.state==='attack' && e.p>=.82) onSwipe(activeTrial?.mirrored?OPP[e.dir]:e.dir);
 },
 afterUpdate() {
  if(window.__scenario==='drift-calm') {
   const state=presentation.environmentState;
   for(let i=state.leaves.length-1;i>=0;i--) if(state.leaves[i].gust) { state.leaves.splice(i,1); state.leafMotion.invalidate(); }
  }
 },
};
